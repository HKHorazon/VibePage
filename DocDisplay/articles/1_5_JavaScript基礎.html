JavaScript：網頁的大腦

2026-05-19

JS 寫在哪裡

寫在 HTML 的 <script> 標籤裡，通常放在 <body> 最底部。

<body>
  <!-- 其他內容 -->

  <script>
    // JavaScript 寫在這裡
    console.log("Hello!")
  </script>
</body>

變數

用來儲存資料。let 可以改變，const 不能改變。

let score = 0          // 數字
let name = "小明"      // 文字
let isAlive = true     // 布林值（true / false）

const MAX_SPEED = 10   // 固定不變的值

條件判斷

「如果...就...否則...」的邏輯。

if (score >= 100) {
  console.log("過關！")
} else if (score >= 50) {
  console.log("繼續加油")
} else {
  console.log("再試一次")
}

函式

把一段程式打包成一個名字，需要時呼叫它。

function jump() {
  bird.y -= 80    // 小鳥往上移動
}

function addScore(points) {
  score += points
  console.log("目前分數：" + score)
}

事件監聽

偵測使用者的操作，再執行對應的函式。

// 點擊按鈕
document.getElementById("startBtn").addEventListener("click", startGame)

// 鍵盤按下空白鍵
document.addEventListener("keydown", function(e) {
  if (e.code === "Space") {
    jump()
  }
})

// 手機觸控
window.addEventListener("touchstart", jump)

操作 HTML 元素

用 JS 抓取頁面上的元素，然後修改它的內容或樣式。

// 抓取元素
let scoreEl = document.getElementById("score")
let box = document.querySelector(".box")

// 修改文字
scoreEl.textContent = "分數：" + score

// 修改樣式
box.style.backgroundColor = "red"
box.style.display = "none"    // 隱藏元素

遊戲迴圈

每秒重複執行 60 次，讓畫面持續更新。

function gameLoop() {
  update()    // 更新所有物件的位置
  draw()      // 重新繪製畫面
  requestAnimationFrame(gameLoop)
}

// 啟動迴圈
requestAnimationFrame(gameLoop)

Canvas 繪圖

在畫布上畫圖形，是遊戲的核心技術。

let canvas = document.getElementById("myCanvas")
let ctx = canvas.getContext("2d")

// 畫一個填滿的矩形
ctx.fillStyle = "#3b82f6"
ctx.fillRect(x, y, width, height)

// 清除畫布（每次更新前先清除）
ctx.clearRect(0, 0, canvas.width, canvas.height)

// 畫圓形
ctx.beginPath()
ctx.arc(x, y, radius, 0, Math.PI * 2)
ctx.fill()
